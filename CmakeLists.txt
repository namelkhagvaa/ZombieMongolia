
cmake_minimum_required(VERSION 3.20)
project(ZombieMongolia LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Build төрөл заагаагүй бол оновчтой (Release) хувилбараар хөрвүүлнэ: том газар нутгийг үүсгэх,
# олон зомби зурахад оновчлолгүй хувилбар хэт удаан. Debug хэрэгтэй бол: cmake -B build -DCMAKE_BUILD_TYPE=Debug
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include(FetchContent)

set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)

FetchContent_Declare(
    raylib
    GIT_REPOSITORY https://github.com/raysan5/raylib.git
    GIT_TAG 5.5
)

FetchContent_MakeAvailable(raylib)

add_executable(ZombieMongolia
    main.cpp
    src/Campaign.cpp
    src/CampaignData.cpp
    src/Environment.cpp
    src/Game.cpp
    src/Grenades.cpp
    src/Hud.cpp
    src/Humanoid.cpp
    src/Layout.cpp
    src/MeshBuilder.cpp
    src/Minimap.cpp
    src/Mission.cpp
    src/Mover.cpp
    src/Particles.cpp
    src/Player.cpp
    src/Polygon.cpp
    src/Procedural.cpp
    src/Progress.cpp
    src/Regions.cpp
    src/Shop.cpp
    src/Soldier.cpp
    src/Structures.cpp
    src/Terrain.cpp
    src/ThirdPersonCamera.cpp
    src/Tracers.cpp
    src/UiFont.cpp
    src/Vehicle.cpp
    src/Weapon.cpp
    src/World.cpp
    src/Zombie.cpp
)
target_include_directories(ZombieMongolia PRIVATE src)
target_link_libraries(ZombieMongolia PRIVATE raylib)

if(NOT MSVC)
    target_compile_options(ZombieMongolia PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
endif()
