# олон зомби зурахад оновчлолгүй хувилбар хэт удаан. Debug хэрэгтэй бол: cmake -B build -DCMAKE_BUILD_TYPE=Debug
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include(FetchContent)

# raylib-ийг Android платформоор, OpenGL ES 3.0-оор хөрвүүлнэ.
if(ANDROID)
    set(PLATFORM "Android" CACHE STRING "" FORCE)
    set(OPENGL_VERSION "ES 3.0" CACHE STRING "" FORCE)
endif()

# Хөтөч (Emscripten-ий toolchain-аар тохируулсан үед; tools\build_web.ps1 үүнийг хийнэ):
# raylib-ийг Web платформоор, OpenGL ES 3.0 (WebGL2)-оор хөрвүүлнэ.
if(EMSCRIPTEN)
    set(PLATFORM "Web" CACHE STRING "" FORCE)
    set(OPENGL_VERSION "ES 3.0" CACHE STRING "" FORCE)
endif()

FetchContent_Declare(
    raylib
    GIT_REPOSITORY https://github.com/raysan5/raylib.git
    GIT_TAG 5.5
)

FetchContent_MakeAvailable(raylib)
if(EMSCRIPTEN)
    # Хөтөч: том текстуруудыг JPG болгож багасгадаг (tools/pack_web.py)
    target_compile_definitions(raylib PRIVATE SUPPORT_FILEFORMAT_JPG=1)
endif()

# raylib-ийн үндсэн vertex shader (DrawTriangle3D, DrawCube, DrawLine3D г.м.) OpenGL ES дээр mediump: олон утасны GPU
# үүнийг 16 битээр тооцдог тул мянган метрийн ертөнцийн координатад ~2 м нарийвчлалтай болж, хөтлөгч сум, сумны мөр
# чичирч хэлбэр нь эвдэрдэг байсан (компьютер үргэлж highp). Vertex shader-т highp заавал дэмжигддэг.
set(RLGL_HEADER ${raylib_SOURCE_DIR}/src/rlgl.h)
file(READ ${RLGL_HEADER} RLGL_TEXT)
set(RLGL_PATCHED "${RLGL_TEXT}")
foreach(api "ES3 (WebGL 2)" "ES2 (WebGL)")
    string(REPLACE "\"precision mediump float;           \\n\"     // Precision required for OpenGL ${api} (on some browsers)"
                   "\"precision highp float;             \\n\"     // Precision required for OpenGL ${api} (on some browsers)"
                   RLGL_PATCHED "${RLGL_PATCHED}")
endforeach()
if(NOT RLGL_PATCHED STREQUAL RLGL_TEXT)
    file(WRITE ${RLGL_HEADER} "${RLGL_PATCHED}")
    message(STATUS "raylib: default vertex shader set to highp")
endif()

if(ANDROID)
    # Android дээр тоглоом нь NativeActivity-ийн ачаалдаг сан (libZombieMongolia.so) болно.
    add_library(ZombieMongolia SHARED ${GAME_SOURCES})
    target_include_directories(ZombieMongolia PRIVATE ${ANDROID_NDK}/sources/android/native_app_glue)
    target_link_options(ZombieMongolia PRIVATE -u ANativeActivity_onCreate)
    target_link_libraries(ZombieMongolia PRIVATE log android EGL GLESv3 m)
else()
    if(WIN32)
        # Exe-ийн икон (assets\app.ico)
        enable_language(RC)
        list(APPEND GAME_SOURCES src/app.rc)
    endif()
    add_executable(ZombieMongolia ${GAME_SOURCES})

target_include_directories(ZombieMongolia PRIVATE src ${enet_SOURCE_DIR}/include ${CMAKE_BINARY_DIR}/generated)
# Хувилбарын цаг (src/BuildStamp.cpp): build бүрд шинэчилнэ
set(BUILD_STAMP_VALUE ${CMAKE_BINARY_DIR}/generated/BuildStampValue.h)
add_custom_target(build_stamp
    COMMAND ${CMAKE_COMMAND} -DOUT=${BUILD_STAMP_VALUE} -P ${CMAKE_SOURCE_DIR}/cmake/BuildStamp.cmake
    BYPRODUCTS ${BUILD_STAMP_VALUE}
    VERBATIM)
add_dependencies(ZombieMongolia build_stamp)
target_link_libraries(ZombieMongolia PRIVATE raylib enet)
if(WIN32)
    target_link_libraries(ZombieMongolia PRIVATE ws2_32 winmm winhttp)
    # Суулгагчаар өөр компьютерт тараахад: MinGW-ийн C++ сангуудыг exe-д шингээнэ (libstdc++, libgcc DLL хэрэггүй),
    # консол цонх гаргахгүй.
    target_link_options(ZombieMongolia PRIVATE -static)
    set_target_properties(ZombieMongolia PROPERTIES WIN32_EXECUTABLE TRUE)
endif()

if(EMSCRIPTEN)
    # Хөтчийн хуудас: ZombieMongolia.html + .js + .wasm. Тоглоомд хэрэгтэй файлууд л (Android-ийн APK-тай ижил;
    # эх .glb загварууд хэт том тул орохгүй) web_assets-д бэлтгэгдэж, tools/build_web.ps1 тэднийг tools/pack_web.py-оор
    # багцална (JPG, OGG болгож; эхлэх, ар талд, аймаг бүрд тусдаа татагдана: web/shell.html).
    set(WEB_ASSETS ${CMAKE_BINARY_DIR}/web_assets)
    file(REMOVE_RECURSE ${WEB_ASSETS})
    file(GLOB WEB_TOP ${CMAKE_SOURCE_DIR}/assets/*_mesh.glb ${CMAKE_SOURCE_DIR}/assets/*_color.png
         ${CMAKE_SOURCE_DIR}/assets/home_back.png ${CMAKE_SOURCE_DIR}/assets/splash.png)
    file(COPY ${WEB_TOP} DESTINATION ${WEB_ASSETS}/assets)
    foreach(sub icons huds terrain)
        file(GLOB files ${CMAKE_SOURCE_DIR}/assets/${sub}/*.png)
        file(COPY ${files} DESTINATION ${WEB_ASSETS}/assets/${sub})
    endforeach()
    foreach(sub cars buu trees pooshig)
        file(GLOB files ${CMAKE_SOURCE_DIR}/assets/${sub}/*.vmesh ${CMAKE_SOURCE_DIR}/assets/${sub}/*_color.png ${CMAKE_SOURCE_DIR}/assets/${sub}/*_far.bin)
        file(COPY ${files} DESTINATION ${WEB_ASSETS}/assets/${sub})
    endforeach()
    file(GLOB files ${CMAKE_SOURCE_DIR}/assets/city/*.bin) # Улаанбаатар, аймгийн төвүүдийн газрын зураг
    file(COPY ${files} DESTINATION ${WEB_ASSETS}/assets/city)
    file(GLOB files ${CMAKE_SOURCE_DIR}/assets/sounds/*.wav)
    file(COPY ${files} DESTINATION ${WEB_ASSETS}/assets/sounds)
    file(COPY ${CMAKE_SOURCE_DIR}/android/ui_font.ttf DESTINATION ${WEB_ASSETS})

    set_target_properties(ZombieMongolia PROPERTIES SUFFIX ".html")
    target_link_options(ZombieMongolia PRIVATE
        -sUSE_GLFW=3
        -sMIN_WEBGL_VERSION=2
        -sMAX_WEBGL_VERSION=2
        -sASYNCIFY                 # үндсэн давталт хэвээрээ: frame бүрд хөтчид удирдлагыг буцаана (Platform.cpp)
        -sASYNCIFY_STACK_SIZE=1048576
        -sALLOW_MEMORY_GROWTH=1
        -sINITIAL_MEMORY=256MB      # утасны Safari том санах ойг эхэнд нь өгч чаддаггүй; хэрэгтэй үед өснө
        -sSTACK_SIZE=8MB
        # HEAPF32: raylib-ийн дууны сан (miniaudio); stringToUTF8, UTF8ToString, lengthBytesUTF8: fetch, localStorage (Online.cpp)
        # FS, addRunDependency, removeRunDependency: багц татагч файлуудыг FS-д бичнэ (web/shell.html)
        -sEXPORTED_RUNTIME_METHODS=HEAPF32,stringToUTF8,UTF8ToString,lengthBytesUTF8,FS,addRunDependency,removeRunDependency
        -sFORCE_FILESYSTEM=1
        --shell-file ${CMAKE_SOURCE_DIR}/web/shell.html
    )
endif()
