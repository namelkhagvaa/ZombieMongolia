
cmake_minimum_required(VERSION 3.20)
project(ZombieMongolia LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Build төрөл заагаагүй бол оновчтой (Release) хувилбараар хөрвүүлнэ: том газар нутгийг үүсгэх,
# олон зомби зурахад оновчлолгүй хувилбар хэт удаан. Debug хэрэгтэй бол: cmake -B build -DCMAKE_BUILD_TYPE=Debug
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include(FetchContent)

set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)

# Android (NDK-ийн toolchain-аар тохируулсан үед; tools\build_android.ps1 үүнийг хийнэ):
# raylib-ийг Android платформоор, OpenGL ES 3.0-оор хөрвүүлнэ.
if(ANDROID)
    set(PLATFORM "Android" CACHE STRING "" FORCE)
    set(OPENGL_VERSION "ES 3.0" CACHE STRING "" FORCE)
endif()

# Хөтөч (Emscripten-ий toolchain-аар тохируулсан үед; tools\build_web.ps1 үүнийг хийнэ):
# raylib-ийг Web платформоор, OpenGL ES 3.0 (WebGL2)-оор хөрвүүлнэ.
if(EMSCRIPTEN)
    set(PLATFORM "Web" CACHE STRING "" FORCE)
    set(OPENGL_VERSION "ES 3.0" CACHE STRING "" FORCE)
endif()

FetchContent_Declare(
    raylib
    GIT_REPOSITORY https://github.com/raysan5/raylib.git
    GIT_TAG 5.5
)

FetchContent_MakeAvailable(raylib)

# Wi-Fi дээрх хамтарсан тоглолт: ENet (UDP, Android ба Windows дээр ажилладаг жижиг сан)
FetchContent_Declare(
    enet
    GIT_REPOSITORY https://github.com/lsalzman/enet.git
    GIT_TAG v1.3.18
)
FetchContent_MakeAvailable(enet)

set(GAME_SOURCES
    main.cpp
    src/Airstrike.cpp
    src/Audio.cpp
    src/Campaign.cpp
    src/Splash.cpp
    src/CampaignData.cpp
    src/CharacterModel.cpp
    src/Environment.cpp
    src/FuelBarrels.cpp
    src/Game.cpp
    src/Grenades.cpp
    src/Hud.cpp
    src/Input.cpp
    src/HorseModel.cpp
    src/Humanoid.cpp
    src/Layout.cpp
    src/MedKits.cpp
    src/MeshBuilder.cpp
    src/Minimap.cpp
    src/Mission.cpp
    src/Mover.cpp
    src/Net.cpp
    src/Particles.cpp
    src/Platform.cpp
    src/Player.cpp
    src/Polygon.cpp
    src/Procedural.cpp
    src/Progress.cpp
    src/Regions.cpp
    src/Shop.cpp
    src/Soldier.cpp
    src/Structures.cpp
    src/Team.cpp
    src/Terrain.cpp
    src/ThirdPersonCamera.cpp
    src/Tracers.cpp
    src/UiFont.cpp
    src/UiStyle.cpp
    src/Vehicle.cpp
    src/VehicleModel.cpp
    src/Weapon.cpp
    src/Weather.cpp
    src/World.cpp
    src/Zombie.cpp
)

if(ANDROID)
    # Android дээр тоглоом нь NativeActivity-ийн ачаалдаг сан (libZombieMongolia.so) болно.
    add_library(ZombieMongolia SHARED ${GAME_SOURCES})
    target_include_directories(ZombieMongolia PRIVATE ${ANDROID_NDK}/sources/android/native_app_glue)
    target_link_options(ZombieMongolia PRIVATE -u ANativeActivity_onCreate)
    target_link_libraries(ZombieMongolia PRIVATE log android EGL GLESv3 m)
else()
    if(WIN32)
        # Exe-ийн икон (assets\app.ico)
        enable_language(RC)
        list(APPEND GAME_SOURCES src/app.rc)
    endif()
    add_executable(ZombieMongolia ${GAME_SOURCES})
endif()
target_include_directories(ZombieMongolia PRIVATE src ${enet_SOURCE_DIR}/include)
target_link_libraries(ZombieMongolia PRIVATE raylib enet)
if(WIN32)
    target_link_libraries(ZombieMongolia PRIVATE ws2_32 winmm)
    # Суулгагчаар өөр компьютерт тараахад: MinGW-ийн C++ сангуудыг exe-д шингээнэ (libstdc++, libgcc DLL хэрэггүй),
    # консол цонх гаргахгүй.
    target_link_options(ZombieMongolia PRIVATE -static)
    set_target_properties(ZombieMongolia PROPERTIES WIN32_EXECUTABLE TRUE)
endif()

if(EMSCRIPTEN)
    # Хөтчийн хуудас: ZombieMongolia.html + .js + .wasm + .data (assets). Тоглоомд хэрэгтэй файлууд л
    # (Android-ийн APK-тай ижил): эх .glb загварууд хэт том тул орохгүй.
    set(WEB_ASSETS ${CMAKE_BINARY_DIR}/web_assets)
    file(REMOVE_RECURSE ${WEB_ASSETS})
    file(GLOB WEB_TOP ${CMAKE_SOURCE_DIR}/assets/*_mesh.glb ${CMAKE_SOURCE_DIR}/assets/*_color.png
         ${CMAKE_SOURCE_DIR}/assets/home_back.png ${CMAKE_SOURCE_DIR}/assets/splash.png)
    file(COPY ${WEB_TOP} DESTINATION ${WEB_ASSETS}/assets)
    foreach(sub icons huds terrain)
        file(GLOB files ${CMAKE_SOURCE_DIR}/assets/${sub}/*.png)
        file(COPY ${files} DESTINATION ${WEB_ASSETS}/assets/${sub})
    endforeach()
    foreach(sub cars buu)
        file(GLOB files ${CMAKE_SOURCE_DIR}/assets/${sub}/*.vmesh ${CMAKE_SOURCE_DIR}/assets/${sub}/*_color.png)
        file(COPY ${files} DESTINATION ${WEB_ASSETS}/assets/${sub})
    endforeach()
    file(GLOB files ${CMAKE_SOURCE_DIR}/assets/sounds/*.wav)
    file(COPY ${files} DESTINATION ${WEB_ASSETS}/assets/sounds)
    file(COPY ${CMAKE_SOURCE_DIR}/android/ui_font.ttf DESTINATION ${WEB_ASSETS})

    set_target_properties(ZombieMongolia PROPERTIES SUFFIX ".html")
    target_link_options(ZombieMongolia PRIVATE
        -sUSE_GLFW=3
        -sMIN_WEBGL_VERSION=2
        -sMAX_WEBGL_VERSION=2
        -sASYNCIFY                 # үндсэн давталт хэвээрээ: frame бүрд хөтчид удирдлагыг буцаана (Platform.cpp)
        -sASYNCIFY_STACK_SIZE=1048576
        -sALLOW_MEMORY_GROWTH=1
        -sINITIAL_MEMORY=256MB      # утасны Safari том санах ойг эхэнд нь өгч чаддаггүй; хэрэгтэй үед өснө
        -sSTACK_SIZE=8MB
        -sEXPORTED_RUNTIME_METHODS=HEAPF32  # raylib-ийн дууны сан (miniaudio) Module.HEAPF32-оор дуу гаргана
        --preload-file ${WEB_ASSETS}@/
        --shell-file ${CMAKE_SOURCE_DIR}/web/shell.html
    )
endif()

if(NOT MSVC)
    target_compile_options(ZombieMongolia PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
endif()
